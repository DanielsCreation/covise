// vis_surfel_color.glslf (Fragment Shader)
#version 420 core

in FS_IN {
    vec3  pass_point_color;
    vec2  pass_uv_coords;
    vec3  pass_world_pos;
    vec3  pass_normal_ws;
    float pass_radius_ws;
    float pass_screen_size;
} fsIn;

layout(location = 0) out vec4 out_color;

#include "vis_color_no_prov.glsl"

// neu: Beleuchtungs-Uniforms
uniform vec3 u_lightPos;
uniform vec3 u_viewPos;
uniform vec3 u_lightColor;
uniform float u_shininess;

// Blinn-Phong-Funktion
vec3 blinn_phong(vec3 normal, vec3 lightDir, vec3 viewDir, vec3 lightColor, float shininess) {
    float diff = max(dot(normal, lightDir), 0.0);
    vec3 halfway = normalize(lightDir + viewDir);
    float spec = pow(max(dot(normal, halfway), 0.0), shininess);
    return lightColor * (diff + spec);
}

void main() {
    if (length(fsIn.pass_uv_coords) > 1.0)
        discard;

    // Basispixel-Farbe
    vec3 baseColor = get_color(
        fsIn.pass_world_pos,
        fsIn.pass_normal_ws,
        fsIn.pass_point_color,
        fsIn.pass_radius_ws,
        fsIn.pass_screen_size
    );

    // Lichtrichtung und Blickrichtung
    vec3 lightDir = normalize(u_lightPos - fsIn.pass_world_pos);
    vec3 viewDir  = normalize(u_viewPos  - fsIn.pass_world_pos);

    // Beleuchtungsberechnung
    vec3 lighting = blinn_phong(normalize(fsIn.pass_normal_ws), lightDir, viewDir, u_lightColor, u_shininess);

    // Endfarbe
    vec3 finalColor = baseColor * lighting;
    out_color = vec4(finalColor, 1.0);
}
