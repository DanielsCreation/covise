bool LamureMeasurement::getLatestAttribute(osg::Stats* stats, const std::string& attributeName, double& value) {
    if (!stats) return false;
    if (stats->getLatestFrameNumber() < stats->getEarliestFrameNumber()) return false;
    return stats->getAttribute(stats->getLatestFrameNumber(), attributeName, value);
}

LamureMeasurement::LamureMeasurement(
    Lamure*                      plugin,
    opencover::VRViewer*         viewer,
    const std::vector<Segment>&  segments,
    const std::string&           logfile)
    : m_plugin(plugin)
    , m_viewer(viewer)
    , m_segments(segments)
    , m_logfile(logfile)
{
    m_running = true;
    m_startTick = osg::Timer::instance()->tick();
    m_lastFrameTick = m_startTick;
    initCallbacks();

    // Get the stats objects
    osg::Stats* viewerStats = m_viewer->getViewerStats();
    osg::Stats* cameraStats = m_viewer->getCamera()->getStats();

    // Explicitly enable the collection of specific metrics
    if (viewerStats) {
        viewerStats->setCollectStats("update", true);
        viewerStats->setCollectStats("sync", true);
        viewerStats->setCollectStats("swap", true);
    }
    if (cameraStats) {
        cameraStats->setCollectStats("rendering", true);
        cameraStats->setCollectGPUMetrics(true);
    }

    std::cout << "Measurement(flight mode) started\n";
}

LamureMeasurement::~LamureMeasurement()
{

}

void LamureMeasurement::stop() {
    m_running = false;
}

void LamureMeasurement::initCallbacks()
{
    auto cam = m_viewer->getCamera();
    m_preCB = new MarkCallback(this, true);
    m_postCB = new MarkCallback(this, false);
    cam->setPreDrawCallback(m_preCB);
    cam->setPostDrawCallback(m_postCB);
}

void LamureMeasurement::drawIncrement(bool preDraw)
{
    if (!m_running) return;

    osg::Timer_t now = osg::Timer::instance()->tick();

    if (preDraw) {
        m_lastFrameTick = now;
    } else {
        FrameStats stats;
        stats.frame_time_ms = osg::Timer::instance()->delta_m(m_lastFrameTick, now);

        const auto& render_info = m_plugin->getRenderInfo();
        stats.rendered_splats = render_info.rendered_splats;
        stats.rendered_nodes = render_info.rendered_nodes;
        stats.rendered_bounding_boxes = render_info.rendered_bounding_boxes;

        osg::Stats* viewerStats = m_viewer->getViewerStats();
        osg::Stats* cameraStats = m_viewer->getCamera()->getStats();
        double value = 0.0;

        // Get stats from the correct object using the latest frame
        if (getLatestAttribute(viewerStats, "Update traversal time taken", value)) stats.cpu_update_ms = value * 1000.0;
        if (getLatestAttribute(viewerStats, "sync time taken", value)) stats.sync_time_ms = value * 1000.0;
        if (getLatestAttribute(viewerStats, "swap time taken", value)) stats.swap_time_ms = value * 1000.0;

        if (cameraStats) {
            if (getLatestAttribute(cameraStats, "Cull traversal time taken", value)) stats.cpu_cull_ms = value * 1000.0;
            if (getLatestAttribute(cameraStats, "Draw traversal time taken", value)) stats.cpu_draw_ms = value * 1000.0;
            if (getLatestAttribute(cameraStats, "GPU time taken", value)) stats.gpu_time_ms = value * 1000.0;
        }

        osg::Matrixd viewMatrix = m_viewer->getCamera()->getViewMatrix();
        stats.position = viewMatrix.getTrans();
        stats.orientation = viewMatrix.getRotate();

        m_stats.push_back(stats);

        double dt = osg::Timer::instance()->delta_s(m_lastFrameTick, now);
        m_segmentTime += dt;

        while (m_currentSegment < m_segments.size())
        {
            const auto& seg = m_segments[m_currentSegment];
            float dist = seg.tra.length();
            double transDur = seg.transSpeed > 0.f ? dist / seg.transSpeed : 0.0;
            float maxAngle = std::max({ std::abs(seg.rot.x()), std::abs(seg.rot.y()), std::abs(seg.rot.z()) });
            double rotDur = seg.rotSpeed > 0.f ? maxAngle / seg.rotSpeed : 0.0;
            double segDuration = std::max(transDur, rotDur);

            if (m_segmentTime < segDuration)
                break;

            m_segmentTime -= segDuration;
            m_cumulativeTra += seg.tra;
            m_cumulativeRot += seg.rot;
            ++m_currentSegment;
        }

        if (m_currentSegment >= m_segments.size())
        {
            m_running = false;
            return;
        }

        const auto& seg = m_segments[m_currentSegment];
        float dist = seg.tra.length();
        double transDur = seg.transSpeed > 0.f ? dist / seg.transSpeed : 0.0;
        float maxAngle = std::max({ std::abs(seg.rot.x()), std::abs(seg.rot.y()), std::abs(seg.rot.z()) });
        double rotDur = seg.rotSpeed > 0.f ? maxAngle / seg.rotSpeed : 0.0;
        double segDuration = std.max(transDur, rotDur);
        double frac = segDuration > 0.0 ? std::min(m_segmentTime / segDuration, 1.0) : 1.0;

        osg::Vec3 tra = m_cumulativeTra + seg.tra * float(frac);
        osg::Vec3 rot = m_cumulativeRot + seg.rot * float(frac);

        updateCamera(tra, rot);
    }
}

void LamureMeasurement::updateCamera(const osg::Vec3& tra, const osg::Vec3& rot)
{
    double rx = osg::DegreesToRadians(rot.x());
    double ry = osg::DegreesToRadians(rot.y());
    double rz = osg::DegreesToRadians(rot.z());

    osg::Matrix mrx, mry, mrz;
    mrx.makeRotate(rx, 1, 0, 0);
    mry.makeRotate(ry, 0, 1, 0);
    mrz.makeRotate(rz, 0, 0, 1);
    osg::Matrix rotMat = mrx * mry * mrz;

    osg::Matrix dcs;
    dcs.postMult(rotMat);

    osg::Vec3 vp = opencover::cover->getViewerMat().getTrans();

    dcs.postMult(osg::Matrix::translate(-vp));
    dcs.postMult(osg::Matrix::translate(tra));
    dcs.postMult(osg::Matrix::translate(vp));

    opencover::cover->setXformMat(dcs);
}

void LamureMeasurement::writeLogAndStop()
{
    if (m_stats.empty())
        return;

    std::string base_path = m_logfile;
    std::string::size_type pos = base_path.find_last_of(".");
    if (pos != std::string::npos) {
        base_path = base_path.substr(0, pos);
    }

    std::string frames_path = base_path + "_frames.csv";
    std::string summary_path = base_path + "_summary.csv";

    // Write frames file
    std::ofstream frames_out(frames_path);
    if (!frames_out) {
        std::cerr << "[Measurement] Fehler beim Öffnen von " << frames_path << "\n";
    } else {
        frames_out.imbue(std::locale::classic());
        frames_out << std::fixed << std::setprecision(4);
        frames_out << "frame_time_ms,rendered_splats,rendered_nodes,rendered_bounding_boxes,cpu_update_ms,cpu_cull_ms,cpu_draw_ms,gpu_time_ms,sync_time_ms,swap_time_ms,pos_x,pos_y,pos_z,quat_x,quat_y,quat_z,quat_w\n";
        for (const auto& s : m_stats) {
            frames_out << s.frame_time_ms << ","
                 << s.rendered_splats << ","
                 << s.rendered_nodes << ","
                 << s.rendered_bounding_boxes << ","
                 << s.cpu_update_ms << ","
                 << s.cpu_cull_ms << ","
                 << s.cpu_draw_ms << ","
                 << s.gpu_time_ms << ","
                 << s.sync_time_ms << ","
                 << s.swap_time_ms << ","
                 << s.position.x() << "," << s.position.y() << "," << s.position.z() << ","
                 << s.orientation.x() << "," << s.orientation.y() << "," << s.orientation.z() << "," << s.orientation.w() << "\n";
        }
    }

    // Write summary file
    std::ofstream summary_out(summary_path);
    if (!summary_out) {
        std::cerr << "[Measurement] Fehler beim Öffnen von " << summary_path << "\n";
    } else {
        summary_out.imbue(std::locale::classic());
        summary_out << std::fixed << std::setprecision(4);

        size_t n = m_stats.size();
        double total_time_ms = 0;
        uint64_t total_splats = 0, total_nodes = 0, total_boxes = 0;
        double total_cpu_update = 0, total_cpu_cull = 0, total_cpu_draw = 0, total_gpu = 0, total_sync = 0, total_swap = 0;

        for (const auto& s : m_stats) {
            total_time_ms += s.frame_time_ms;
            total_splats += s.rendered_splats;
            total_nodes += s.rendered_nodes;
            total_boxes += s.rendered_bounding_boxes;
            total_cpu_update += s.cpu_update_ms;
            total_cpu_cull += s.cpu_cull_ms;
            total_cpu_draw += s.cpu_draw_ms;
            total_gpu += s.gpu_time_ms;
            total_sync += s.sync_time_ms;
            total_swap += s.swap_time_ms;
        }

        double avg_frame_time = total_time_ms / n;
        double avg_fps = 1000.0 / avg_frame_time;

        summary_out << "Metric,Value\n";
        summary_out << "Total Frames," << n << "\n";
        summary_out << "Total Time (s)," << total_time_ms / 1000.0 << "\n";
        summary_out << "Avg Frame Time (ms)," << avg_frame_time << "\n";
        summary_out << "Avg FPS," << avg_fps << "\n";
        summary_out << "Avg Splats/Frame," << total_splats / n << "\n";
        summary_out << "Avg Nodes/Frame," << total_nodes / n << "\n";
        summary_out << "Avg Boxes/Frame," << total_boxes / n << "\n";
        summary_out << "Avg CPU Update (ms)," << total_cpu_update / n << "\n";
        summary_out << "Avg CPU Cull (ms)," << total_cpu_cull / n << "\n";
        summary_out << "Avg CPU Draw (ms)," << total_cpu_draw / n << "\n";
        summary_out << "Avg GPU Time (ms)," << total_gpu / n << "\n";
        summary_out << "Avg Sync Time (ms)," << total_sync / n << "\n";
        summary_out << "Avg Swap Time (ms)," << total_swap / n << "\n";
    }
    std::cout << "Measurement written to: " << summary_path << "\n" << frames_path << "\n";
    m_stats.clear();
    m_segmentTime = 0.0;
    m_currentSegment = 0;
    m_cumulativeRot = osg::Vec3(0.0f, 0.0f, 0.0f);
}
